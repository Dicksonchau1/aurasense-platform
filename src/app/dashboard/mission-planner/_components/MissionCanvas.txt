"use client";

import { useEffect, useRef, useState } from "react";

export type Mode = "add" | "move" | "delete";

export interface Waypoint {
  x: number;
  y: number;
  alt: number;
}

export interface Building {
  x: number;
  y: number;
  w: number;
  h: number;
  label: string;
}

export interface MissionStats {
  count: number;
  distance: number;
  duration: number;
  battery: number;
}

interface Props {
  width?: number;
  height?: number;
  onStatsChange?: (s: MissionStats) => void;
  onSim?: (running: boolean) => void;
}

interface Pt {
  x: number;
  y: number;
}

const HOME: Pt = { x: 80, y: 520 };

const BUILDINGS: Building[] = [
  { x: 180, y: 200, w: 90, h: 220, label: "Tower A" },
  { x: 320, y: 260, w: 70, h: 180, label: "Bldg B" },
  { x: 460, y: 160, w: 110, h: 280, label: "Tower C" },
  { x: 640, y: 220, w: 80, h: 200, label: "Bldg D" },
  { x: 760, y: 300, w: 100, h: 140, label: "Annex E" },
];

function dist(a: Pt, b: Pt): number {
  return Math.hypot(a.x - b.x, a.y - b.y);
}

export default function MissionCanvas({
  width = 980,
  height = 600,
  onStatsChange,
  onSim,
}: Props) {
  const canvasRef = useRef<HTMLCanvasElement>(null);
  const [waypoints, setWaypoints] = useState<Waypoint[]>([]);
  const [mode, setMode] = useState<Mode>("add");
  const [sim, setSim] = useState(false);
  const [t, setT] = useState(0);
  const trail = useRef<Array<{ x: number; y: number; a: number }>>([]);

  const stats: MissionStats = (() => {
    if (waypoints.length === 0) {
      return { count: 0, distance: 0, duration: 0, battery: 100 };
    }
    let d = dist(HOME, waypoints[0]);
    for (let i = 1; i < waypoints.length; i++) {
      d += dist(waypoints[i - 1], waypoints[i]);
    }
    d += dist(waypoints[waypoints.length - 1], HOME);
    const meters = d * 0.6;
    const speed = 6;
    const duration = meters / speed;
    const battery = Math.max(0, 100 - (duration / 60) * 2.4);
    return {
      count: waypoints.length,
      distance: meters,
      duration,
      battery,
    };
  })();

  useEffect(() => {
    onStatsChange?.(stats);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [waypoints.length, stats.distance]);

  useEffect(() => {
    onSim?.(sim);
  }, [sim, onSim]);

  useEffect(() => {
    if (!sim || waypoints.length === 0) return;
    let raf = 0;
    let last = performance.now();
    const total = 4500;
    const step = (now: number) => {
      const dt = now - last;
      last = now;
      setT((prev) => {
        const next = prev + dt / total;
        if (next >= 1) {
          setSim(false);
          return 0;
        }
        return next;
      });
      raf = requestAnimationFrame(step);
    };
    raf = requestAnimationFrame(step);
    return () => cancelAnimationFrame(raf);
  }, [sim, waypoints.length]);

  useEffect(() => {
    if (!sim) trail.current = [];
  }, [sim]);

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const ctx = canvas.getContext("2d");
    if (!ctx) return;
    drawMP(ctx, canvas.width, canvas.height, waypoints, sim, t, trail.current);
  }, [waypoints, sim, t]);

  const onPointerDown = (e: React.PointerEvent<HTMLCanvasElement>) => {
    const canvas = canvasRef.current;
    if (!canvas) return;
    const rect = canvas.getBoundingClientRect();
    const x = ((e.clientX - rect.left) / rect.width) * canvas.width;
    const y = ((e.clientY - rect.top) / rect.height) * canvas.height;

    if (mode === "add") {
      setWaypoints((wps) => [...wps, { x, y, alt: 50 + wps.length * 5 }]);
    } else if (mode === "delete") {
      setWaypoints((wps) => {
        let best = -1;
        let bestD = 18;
        wps.forEach((w, i) => {
          const d = Math.hypot(w.x - x, w.y - y);
          if (d < bestD) {
            bestD = d;
            best = i;
          }
        });
        return best >= 0 ? wps.filter((_, i) => i !== best) : wps;
      });
    } else if (mode === "move" && waypoints.length > 0) {
      setWaypoints((wps) => {
        let best = -1;
        let bestD = 18;
        wps.forEach((w, i) => {
          const d = Math.hypot(w.x - x, w.y - y);
          if (d < bestD) {
            bestD = d;
            best = i;
          }
        });
        if (best < 0) return wps;
        return wps.map((w, i) => (i === best ? { ...w, x, y } : w));
      });
    }
  };

  const undo = () => setWaypoints((wps) => wps.slice(0, -1));
  const clearAll = () => {
    setWaypoints([]);
    setSim(false);
    setT(0);
  };
  const toggleSim = () => {
    if (waypoints.length === 0) return;
    setSim((s) => !s);
    setT(0);
  };

  return (
    <div>
      <div
        style={{
          display: "flex",
          gap: 8,
          alignItems: "center",
          marginBottom: 10,
          flexWrap: "wrap",
        }}
      >
        <ModeBtn label="Add" active={mode === "add"} onClick={() => setMode("add")} />
        <ModeBtn label="Move" active={mode === "move"} onClick={() => setMode("move")} />
        <ModeBtn label="Delete" active={mode === "delete"} onClick={() => setMode("delete")} />
        <div style={{ width: 1, height: 22, background: "#1a1f26", margin: "0 4px" }} />
        <SmallBtn onClick={undo} disabled={waypoints.length === 0}>
          Undo
        </SmallBtn>
        <SmallBtn onClick={clearAll} disabled={waypoints.length === 0}>
          Clear
        </SmallBtn>
        <SmallBtn onClick={toggleSim} disabled={waypoints.length === 0} variant="primary">
          {sim ? "Stop" : "Simulate"}
        </SmallBtn>
        <div style={{ marginLeft: "auto", fontSize: 11, opacity: 0.6 }}>
          {waypoints.length} waypoint{waypoints.length === 1 ? "" : "s"} ·{" "}
          {stats.distance.toFixed(0)} m · {stats.duration.toFixed(0)} s ·{" "}
          {stats.battery.toFixed(0)}% remaining
        </div>
      </div>

      anvas
        ref={canvasRef}
        width={width}
        height={height}
        onPointerDown={onPointerDown}
        style={{
          width: "100%",
          height: "auto",
          maxHeight: "70vh",
          display: "block",
          background: "#0a0e15",
          border: "1px solid #1a1f26",
          borderRadius: 8,
          cursor:
            mode === "add"
              ? "crosshair"
              : mode === "delete"
              ? "not-allowed"
              : "grab",
        }}
      />
    </div>
  );
}

function ModeBtn({
  label,
  active,
  onClick,
}: {
  label: string;
  active: boolean;
  onClick: () => void;
}) {
  return (
    <button
      onClick={onClick}
      style={{
        padding: "6px 12px",
        borderRadius: 6,
        background: active ? "#22d3ee" : "#11151a",
        color: active ? "#0b0d10" : "#e7ecf3",
        border: "1px solid #1a1f26",
        fontSize: 12,
        fontWeight: 700,
        cursor: "pointer",
      }}
    >
      {label}
    </button>
  );
}

function SmallBtn({
  children,
  onClick,
  disabled,
  variant,
}: {
  children: React.ReactNode;
  onClick: () => void;
  disabled?: boolean;
  variant?: "primary";
}) {
  const palette =
    variant === "primary"
      ? { bg: "#22d3ee", fg: "#0b0d10" }
      : { bg: "#11151a", fg: "#e7ecf3" };
  return (
    <button
      onClick={onClick}
      disabled={disabled}
      style={{
        padding: "6px 12px",
        borderRadius: 6,
        background: disabled ? "#1a1f26" : palette.bg,
        color: disabled ? "#6b7280" : palette.fg,
        border: "1px solid #1a1f26",
        fontSize: 12,
        fontWeight: 600,
        cursor: disabled ? "default" : "pointer",
      }}
    >
      {children}
    </button>
  );
}

function drawMP(
  ctx: CanvasRenderingContext2D,
  W: number,
  H: number,
  waypoints: Waypoint[],
  sim: boolean,
  t: number,
  trail: Array<{ x: number; y: number; a: number }>,
) {
  ctx.fillStyle = "#0a0e15";
  ctx.fillRect(0, 0, W, H);

  ctx.strokeStyle = "#11151a";
  ctx.lineWidth = 1;
  for (let x = 0; x <= W; x += 40) {
    ctx.beginPath();
    ctx.moveTo(x + 0.5, 0);
    ctx.lineTo(x + 0.5, H);
    ctx.stroke();
  }
    for (let y = 0; y <= H; y += 40) {
    ctx.beginPath();
    ctx.moveTo(0, y + 0.5);
    ctx.lineTo(W, y + 0.5);
    ctx.stroke();
  }

  ctx.fillStyle = "#1a1f26";
  ctx.font = "10px ui-monospace, Menlo, monospace";
  ctx.fillText("N", W - 18, 14);
  ctx.fillText("100 m", 16, H - 10);
  ctx.strokeStyle = "#1a1f26";
  ctx.beginPath();
  ctx.moveTo(12, H - 18);
  ctx.lineTo(78, H - 18);
  ctx.stroke();

  for (const b of BUILDINGS) {
    ctx.fillStyle = "rgba(0,0,0,0.45)";
    ctx.fillRect(b.x + 4, b.y + 6, b.w, b.h);
    const grad = ctx.createLinearGradient(b.x, b.y, b.x, b.y + b.h);
    grad.addColorStop(0, "#1b232c");
    grad.addColorStop(1, "#0f1419");
    ctx.fillStyle = grad;
    ctx.fillRect(b.x, b.y, b.w, b.h);
    ctx.strokeStyle = "#2a323d";
    ctx.lineWidth = 1;
    ctx.strokeRect(b.x + 0.5, b.y + 0.5, b.w, b.h);
    ctx.fillStyle = "rgba(34,211,238,0.08)";
    for (let yy = b.y + 10; yy < b.y + b.h - 6; yy += 18) {
      for (let xx = b.x + 6; xx < b.x + b.w - 6; xx += 14) {
        ctx.fillRect(xx, yy, 8, 10);
      }
    }
    ctx.fillStyle = "#9ca3af";
    ctx.font = "10px ui-monospace, Menlo, monospace";
    ctx.fillText(b.label, b.x + 4, b.y - 4);
  }

  ctx.save();
  ctx.translate(HOME.x, HOME.y);
  ctx.fillStyle = "#22c55e";
  ctx.beginPath();
  ctx.arc(0, 0, 9, 0, Math.PI * 2);
  ctx.fill();
  ctx.strokeStyle = "rgba(34,197,94,0.35)";
  ctx.lineWidth = 6;
  ctx.beginPath();
  ctx.arc(0, 0, 14, 0, Math.PI * 2);
  ctx.stroke();
  ctx.fillStyle = "#0b0d10";
  ctx.font = "bold 10px ui-monospace, Menlo, monospace";
  ctx.fillText("H", -3, 4);
  ctx.restore();

  if (waypoints.length > 0) {
    ctx.strokeStyle = "#22d3ee";
    ctx.lineWidth = 2;
    ctx.setLineDash([6, 4]);
    ctx.beginPath();
    ctx.moveTo(HOME.x, HOME.y);
    for (const w of waypoints) ctx.lineTo(w.x, w.y);
    ctx.lineTo(HOME.x, HOME.y);
    ctx.stroke();
    ctx.setLineDash([]);
  }

  waypoints.forEach((w, i) => {
    ctx.fillStyle = "rgba(34,211,238,0.18)";
    ctx.beginPath();
    ctx.arc(w.x, w.y, 12, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "#22d3ee";
    ctx.beginPath();
    ctx.arc(w.x, w.y, 6, 0, Math.PI * 2);
    ctx.fill();
    ctx.fillStyle = "#e7ecf3";
    ctx.font = "bold 10px ui-monospace, Menlo, monospace";
    ctx.fillText("WP" + (i + 1), w.x + 9, w.y - 9);
    ctx.fillStyle = "#9ca3af";
    ctx.font = "10px ui-monospace, Menlo, monospace";
    ctx.fillText(w.alt + "m", w.x + 9, w.y + 4);
  });

  if (sim && waypoints.length > 0) {
    const segs: Array<{ a: Pt; b: Pt }> = [];
    segs.push({ a: HOME, b: waypoints[0] });
    for (let i = 1; i < waypoints.length; i++) {
      segs.push({ a: waypoints[i - 1], b: waypoints[i] });
    }
    segs.push({ a: waypoints[waypoints.length - 1], b: HOME });

    const lens = segs.map((s) => Math.hypot(s.b.x - s.a.x, s.b.y - s.a.y));
    const total = lens.reduce((s, x) => s + x, 0) || 1;
    const target = total * t;

    let acc = 0;
    let pos: Pt = { x: HOME.x, y: HOME.y };
    for (let i = 0; i < segs.length; i++) {
      if (acc + lens[i] >= target) {
        const localT = (target - acc) / Math.max(1, lens[i]);
        pos = {
          x: segs[i].a.x + (segs[i].b.x - segs[i].a.x) * localT,
          y: segs[i].a.y + (segs[i].b.y - segs[i].a.y) * localT,
        };
        break;
      }
      acc += lens[i];
    }

    trail.push({ x: pos.x, y: pos.y, a: 1 });
    while (trail.length > 80) trail.shift();
    for (const p of trail) p.a *= 0.965;

    for (const p of trail) {
      ctx.fillStyle = "rgba(34,211,238," + (p.a * 0.45).toFixed(3) + ")";
      ctx.beginPath();
      ctx.arc(p.x, p.y, 4, 0, Math.PI * 2);
      ctx.fill();
    }

    const glow = ctx.createRadialGradient(pos.x, pos.y, 0, pos.x, pos.y, 28);
    glow.addColorStop(0, "rgba(34,211,238,0.6)");
    glow.addColorStop(1, "rgba(34,211,238,0)");
    ctx.fillStyle = glow;
    ctx.beginPath();
    ctx.arc(pos.x, pos.y, 28, 0, Math.PI * 2);
    ctx.fill();

    ctx.fillStyle = "#e7ecf3";
    ctx.beginPath();
    ctx.arc(pos.x, pos.y, 5, 0, Math.PI * 2);
    ctx.fill();
    ctx.strokeStyle = "#22d3ee";
    ctx.lineWidth = 2;
    ctx.beginPath();
    ctx.arc(pos.x, pos.y, 7, 0, Math.PI * 2);
    ctx.stroke();
  }
}
